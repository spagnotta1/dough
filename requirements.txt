# Pinned to the versions the test suite is verified green against.
#
# These were previously wrong -- the file claimed scikit-learn 1.3.2, pandas
# 2.2.1, Werkzeug 3.0.1 and python-dateutil 2.8.2 while the working environment
# had 1.7.0, 2.1.3, 3.1.3 and 2.9.0. CI installs from this file, so a stale pin
# means CI tests a dependency set nobody runs. Corrected to match reality rather
# than reinstalling the environment to match the file, because the environment
# is the one with 194 passing tests.
#
# Upgrades are a deliberate change with their own test run, not a side effect of
# an unpinned install.

# --- Web framework ---
# Flask 3.1.3: below it, reading a session key with `in` skipped the
# `Vary: Cookie` header, so a caching proxy could serve one user's page to
# another (PYSEC-2026-2151).
# Werkzeug 3.1.6: `safe_join` accepted Windows device names (`CON`, `NUL`,
# including `example/NUL` and `CON.txt.html`), and `send_from_directory`
# serves static files through it — a GET could hang the server forever. This
# app is developed and run on Windows, so it was directly exposed.
Flask==3.1.3
Werkzeug==3.1.6
Flask-SQLAlchemy==3.1.1
Flask-Migrate==4.0.5
SQLAlchemy==2.0.27

# --- Auth ---
# argon2id password hashing. RFC 9106 profiles require >= 21.2.0.
argon2-cffi==21.3.0

# --- Data / analysis ---
pandas==2.1.3
python-dateutil==2.9.0.post0
# IsolationForest, used for transaction anomaly scoring.
scikit-learn==1.7.0

# --- Integrations ---
anthropic==0.87.0
requests==2.33.0
# Fernet, for encrypting stored institution tokens at rest, and the
# self-signed dev certificate Plaid's HTTPS redirect requirement forces.
#
# 41.0.7 -> 48.0.1 is the one upgrade here that could have destroyed data
# rather than just breaking a test: it holds the key to every stored bank
# token. Verified before pinning -- all four `auth_blob` values in the live
# database still decrypt, and a fresh round-trip still reads back. Fernet's
# on-disk format is unchanged; the CVEs were in the statically linked OpenSSL
# and in EC/PKCS12 parsing paths this app never reaches.
cryptography==48.0.1

# --- Utilities ---
python-dotenv==1.2.2
Markdown==3.8.2
# Icon rasterisation in tools/build_icons.py. Was an undeclared dependency of
# that script -- it imported Pillow without the file ever asking for it.
# None of its many image-parsing CVEs are reachable from the running app: it
# is a build-time tool fed the repo's own artwork, and no request path decodes
# an image. Kept current anyway, because "unreachable today" is a property of
# the code as it stands, not a guarantee about the next feature.
pillow==12.3.0

# --- Serving ---
# The production WSGI server. Werkzeug's built-in server is single-threaded and
# explicitly not for production; `app.run` in app.py's __main__ block is a
# development convenience and nothing deploys through it.
#
# Run with one worker and several threads, never several workers -- see
# Procfile and docs/deploy-railway.md. Three parts of this application keep
# state in the serving process and a second worker silently duplicates all of
# it.
gunicorn==26.0.0

# --- Error monitoring ---
# Optional at runtime: `dough/monitoring.py` no-ops when SENTRY_DSN is unset and
# warns (rather than failing to boot) when the DSN is set and this is missing.
# Pinned here anyway so the deployment that *does* set a DSN gets a version the
# suite has run against.
#
# Everything this sends is filtered by `before_send`, which removes stack-frame
# local variables and request bodies wholesale rather than by key name -- a
# frame in this application can hold a decrypted institution token or a
# household's transactions, and no name filter catches that reliably.
sentry-sdk==2.44.0

# --- Tests ---
# 9.x is a major bump; the suite (921 tests, browser included) was run against
# it before this pin moved.
pytest==9.0.3
